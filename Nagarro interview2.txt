CREATE PROCEDURE sp_InsertAndPivotData
    @SourceTable NVARCHAR(128),
    @ColumnToPivot NVARCHAR(128),
    @ValueColumn NVARCHAR(128)
AS
BEGIN
    SET NOCOUNT ON;
 
    DECLARE @cols NVARCHAR(MAX), @query NVARCHAR(MAX);
 
    -- 1. Insert Data (example - assuming data is already in a source table)
    -- This section would contain your actual INSERT statements
    -- For demonstration, we'll assume @SourceTable already has data.
 
    -- 2. Build the dynamic pivot query
    SELECT @cols = STUFF((SELECT DISTINCT ',' + QUOTENAME(CAST(T.val AS NVARCHAR(MAX)))
                          FROM (SELECT DISTINCT @ColumnToPivot as val FROM @SourceTable) AS T
                          FOR XML PATH(''), TYPE
                         ).value('.', 'NVARCHAR(MAX)') 
                        ,1,1,'');
 
    SET @query = 'SELECT ' + @ColumnToPivot + ', ' + @cols + '
                  FROM 
                  (
                      SELECT ' + @ColumnToPivot + ', ' + @ValueColumn + '
                      FROM ' + @SourceTable + '
                  ) AS SourceTable
                  PIVOT 
                  (
                      SUM(' + @ValueColumn + ') 
                      FOR ' + @ColumnToPivot + ' IN (' + @cols + ')
                  ) AS PivotTable;';
 
    EXECUTE(@query);
END;
