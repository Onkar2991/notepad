public class BasicUser extends User{
private final double discountPercentage = 5;

@Override
public abstract double applyMembershipDiscount(double totalAmount){
return totalAmount-(totalAmount * discountPercentage/100);
}
}




public class PremiumUser extends User{
private final double discountPercentage = 25;

@Override
public abstract double applyMembershipDiscount(double totalAmount){

}
}


public class VipUser extends User{
